## 1. 项目概述

### 1.1 项目定位

本服务为一个通用、可扩展的任务管理后台服务，支持多租户、多用户协作的任务管理场景，可被各种前端应用（Web、移动端、桌面端）调用。

### 1.2 适用场景

- 个人任务管理
- 团队任务协作
- 项目任务追踪
- 工作流管理
- 客服工单系统

### 1.3 核心价值

| 价值点 | 描述 |
|--------|------|
| 多端统一 | 一套服务支持多种前端应用 |
| 实时协作 | 多人同时协作，实时同步状态 |
| 离线支持 | 支持离线操作，自动同步恢复 |
| 高可用性 | 分布式架构，99.9% 可用性 |
| 可扩展性 | 模块化设计，灵活扩展功能 |

### 3.2 核心数据模型

```go
// 组织
type Organization struct {
    ID          uint      `gorm:"primarykey"`
    Name        string    `gorm:"size:100;not null"`
    Code        string    `gorm:"size:50;uniqueIndex;not null"` // 组织代码
    Description string    `gorm:"size:500"`
    Logo        string    `gorm:"size:255"`
    Domain      string    `gorm:"size:100"` // 自定义域名
    Plan        string    `gorm:"size:20;default:free"` // 套餐
    MaxUsers    int       `gorm:"default:10"` // 最大用户数
    MaxProjects int       `gorm:"default:20"` // 最大项目数
    Settings    string    `gorm:"type:text"` // JSON: 组织设置
    Status      string    `gorm:"size:20;default:active"` // active/suspended/deleted
    CreatedAt   time.Time
    UpdatedAt   time.Time
    DeletedAt   gorm.DeletedAt `gorm:"index"`
}

// 用户
type User struct {
    ID            uint      `gorm:"primarykey"`
    Username      string    `gorm:"uniqueIndex;size:50;not null"`
    Email         string    `gorm:"uniqueIndex;size:100;not null"`
    Phone         string    `gorm:"size:20"`
    PasswordHash  string    `gorm:"size:255;not null"`
    Nickname      string    `gorm:"size:50"`
    Avatar        string    `gorm:"size:255"`
    Status        string    `gorm:"size:20;default:active"` // active/inactive/banned
    IsVerified    bool      `gorm:"default:false"`
    LastLoginAt   *time.Time
    LastLoginIP   string    `gorm:"size:50"`
    TwoFactorAuth bool     `gorm:"default:false"`
    Settings      string    `gorm:"type:text"` // JSON: 用户设置
    CreatedAt     time.Time
    UpdatedAt     time.Time
    DeletedAt     gorm.DeletedAt `gorm:"index"`

    ThirdParties []ThirdPartyBinding `gorm:"foreignKey:UserID"`
}

// 第三方绑定
type ThirdPartyBinding struct {
    ID        uint      `gorm:"primarykey"`
    UserID    uint      `gorm:"index;not null"`
    User      User
    Provider  string    `gorm:"size:50;not null;index"` // wechat/github/google
    OpenID    string    `gorm:"size:255;not null;index"` // 第三方用户ID
    UnionID   string    `gorm:"size:255;index"`
    Nickname  string    `gorm:"size:100"`
    Avatar    string    `gorm:"size:255"`
    Token     string    `gorm:"type:text"` // JSON: 访问令牌
    CreatedAt time.Time
    UpdatedAt time.Time
}

// 组织成员
type Member struct {
    ID           uint      `gorm:"primarykey"`
    OrganizationID uint      `gorm:"index;not null"`
    Organization Organization
    UserID       uint      `gorm:"index;not null"`
    User         User
    DepartmentID *uint     `gorm:"index"`
    Department   Department
    Role         string    `gorm:"size:20;default:member"` // owner/admin/member/guest
    Position     string    `gorm:"size:100"`
    Status       string    `gorm:"size:20;default:active"` // active/inactive/pending
    InvitedBy    *uint
    JoinedAt     time.Time
    UpdatedAt    time.Time
    DeletedAt    gorm.DeletedAt `gorm:"index"`
}

// 部门
type Department struct {
    ID             uint      `gorm:"primarykey"`
    OrganizationID uint      `gorm:"index;not null"`
    Organization Organization
    ParentID       *uint     `gorm:"index"`
    Parent         Department
    Name           string    `gorm:"size:100;not null"`
    Description    string    `gorm:"size:500"`
    ManagerID      *uint     `gorm:"index"`
    SortOrder      int       `gorm:"default:0"`
    Status         string    `gorm:"size:20;default:active"`
    CreatedAt      time.Time
    UpdatedAt      time.Time
    DeletedAt      gorm.DeletedAt `gorm:"index"`
}

// 项目
type Project struct {
    ID              uint      `gorm:"primarykey"`
    OrganizationID  uint      `gorm:"index;not null"`
    Organization   Organization
    Name            string    `gorm:"size:100;not null"`
    Code            string    `gorm:"size:50;uniqueIndex;not null"` // 项目代码
    Description     string    `gorm:"type:text"`
    Color           string    `gorm:"size:20;default:#1976d2"`
    Icon            string    `gorm:"size:50"`
    OwnerID         uint      `gorm:"index;not null"`
    Owner           User
    Status          string    `gorm:"size:20;default:active"` // active/archived/deleted
    Visibility      string    `gorm:"size:20;default:private"` // private/organization/public
    Settings        string    `gorm:"type:text"` // JSON: 项目设置
    StartDate       *time.Time
    EndDate         *time.Time
    CreatedAt       time.Time
    UpdatedAt       time.Time
    DeletedAt       gorm.DeletedAt `gorm:"index"`

    Members   []ProjectMember `gorm:"foreignKey:ProjectID"`
    TaskLists []TaskList      `gorm:"foreignKey:ProjectID"`
}

// 项目成员
type ProjectMember struct {
    ID         uint      `gorm:"primarykey"`
    ProjectID  uint      `gorm:"index;not null"`
    Project    Project
    UserID     uint      `gorm:"index;not null"`
    User       User
    Role       string    `gorm:"size:20;default:member"` // owner/admin/member/viewer
    JoinedAt   time.Time
    UpdatedAt  time.Time

    gorm:"uniqueIndex:project_user"
}

// 任务列表（看板）
type TaskList struct {
    ID         uint      `gorm:"primarykey"`
    ProjectID  uint      `gorm:"index;not null"`
    Project    Project
    Name       string    `gorm:"size:100;not null"`
    Color      string    `gorm:"size:20;default:#ffffff"`
    Icon       string    `gorm:"size:50"`
    SortOrder  int       `gorm:"default:0"`
    CreatedAt  time.Time
    UpdatedAt  time.Time
    DeletedAt  gorm.DeletedAt `gorm:"index"`

    Tasks []Task `gorm:"foreignKey:ListID"`
}

// 任务
type Task struct {
    ID           uint      `gorm:"primarykey"`
    OrganizationID uint      `gorm:"index;not null"`
    Organization  Organization
    ProjectID    *uint     `gorm:"index"`
    Project      Project
    TaskListID   *uint     `gorm:"index"`
    TaskList     TaskList
    ParentTaskID *uint     `gorm:"index"` // 父任务ID
    Title        string    `gorm:"size:200;not null"`
    Description  string    `gorm:"type:text"`
    Content      string    `gorm:"type:text"` // 富文本内容
    Status       string    `gorm:"size:20;default:todo"` // todo/in_progress/done/cancelled
    Priority     string    `gorm:"size:20;default:medium"` // low/medium/high/critical
    Position     int       `gorm:"default:0"` // 在列表中的位置
    AssigneeID   *uint     `gorm:"index"` // 负责人
    Assignee     User
    CreatorID    uint      `gorm:"index;not null"`
    Creator      User
    StartDate    *time.Time
    DueDate      *time.Time
    CompletedAt  *time.Time
    EstimatedHours *float64
    ActualHours  *float64
    Progress     int       `gorm:"default:0"` // 0-100
    IsRecurring  bool      `gorm:"default:false"`
    RecurringRule string    `gorm:"size:50"` // daily/weekly/monthly
    RecurringEnd *time.Time
    Watchers    []User    `gorm:"many2many:task_watchers;"`
    Followers   []User    `gorm:"many2many:task_followers;"`
    Settings     string    `gorm:"type:text"` // JSON: 任务设置
    CreatedAt    time.Time
    UpdatedAt    time.Time
    CompletedAt  *time.Time `gorm:"index"`
    DeletedAt    gorm.DeletedAt `gorm:"index"`

    Comments    []TaskComment      `gorm:"foreignKey:TaskID"`
    Attachments []TaskAttachment   `gorm:"foreignKey:TaskID"`
    SubTasks   []Task             `gorm:"foreignKey:ParentTaskID"`
    Dependencies []TaskDependency `gorm:"foreignKey:TaskID"`
    Tags       []Tag             `gorm:"many2many:task_tags;"`
}

// 子任务
type Task struct {
    // ... 同上

type TaskAttachment struct {

    FileType     string    `gorm:"size:50"`
```

## 6. 同步机制

### 6.1 增量同步

**客户端拉取（Pull）**：

```
GET /api/v1/sync/pull?since=1234567890

Response:
{
  "tasks": [...],           // 新增/修改的任务
  "deleted": [...],         // 删除的任务ID
  "timestamp": 1234567900
}
```

**客户端推送（Push）**：

```
POST /api/v1/sync/push

Request Body:
{
  "operations": [
    {
      "type": "create",
      "entity": "task",
      "data": {...}
    },
    {
      "type": "update",
      "entity": "task",
      "id": 123,
      "data": {...}
    }
  ],
  "timestamp": 1234567890
}
```

### 6.2 冲突解决策略

| 冲突类型 | 解决策略 |
|----------|----------|
| 同一任务同时编辑 | 最后更新时间胜出 |
| 任务已被删除 | 删除操作优先 |
| 版本号冲突 | 返回冲突，用户选择保留版本 |

### 6.3 离线支持

```
│  (操作)   │   操作写入  │(队列表)  │   批量同步  │  (处理)   │
└──────────┘             └──────────┘             └──────────┘
```

```yaml

    image: task-server:latest

      - "8080:8080"

    volumes:
      - postgres-data:/var/lib/postgresql/data
    restart: unless-stopped

    image: redis:7-alpine
    command: redis-server --appendonly yes
    volumes:
      - redis-data:/data
    restart: unless-stopped

  minio:
    image: minio/minio:latest
    ports:
      - "9000:9000"
      - "9001:9001"
    environment:
      MINIO_ROOT_USER: minioadmin
      MINIO_ROOT_PASSWORD: ${MINIO_PASSWORD}
    volumes:
      - minio-data:/data
    restart: unless-stopped

    image: prom/prometheus:latest
    ports:
      - "9090:9090"
    volumes:
      - ./prometheus.yml:/etc/prometheus/prometheus.yml
      - prometheus-data:/prometheus
    command:
      - '--config.file=/etc/prometheus/prometheus.yml'
    restart: unless-stopped

      - "3000:3000"
```
